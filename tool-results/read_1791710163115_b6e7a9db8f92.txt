   184→   184→    private fun connect() {
   185→   185→        val url = if (mode == "gateway") "http://$host:$port" else "http://$host:$bridgePort"
   186→   186→        try {
   187→   187→            val opts = IO.Options()
   188→   188→            opts.reconnection = true
   189→   189→            opts.reconnectionDelay = 2_000
   190→   190→            opts.reconnectionDelayMax = 15_000
   191→   191→            opts.timeout = 10_000
   192→   192→            if (mode == "gateway") {
   193→   193→                // نفس نمط عميل الويب: Caddy يوجّه ?XTransformPort إلى منفذ الجسر
   194→   194→                opts.query = "XTransformPort=$bridgePort"
   195→   195→            }
   196→   196→
   197→   197→            val s = IO.socket(url, opts)
   198→   198→            socket = s
   199→   199→
   200→   200→            s.on(Socket.EVENT_CONNECT, onConnect)
   201→   201→            s.on(Socket.EVENT_DISCONNECT) {
   202→   202→                setState("offline")
   203→   203→                // انقطاع القناة = نهاية الجلسة دائمًا (من يتحكم لم يعد متصلًا)
   204→   204→                main.post { endControlSession("disconnected") }
   205→   205→            }
   206→   206→            s.on(Socket.EVENT_CONNECT_ERROR) { setState("reconnecting") }
   207→   207→
   208→   208→            // أوامر الكمبيوتر → الهاتف: إقرار الاستلام (التنفيذ الكامل في مرحلة الحماية)
   209→   209→            for (cmd in AgentBridge.PC_COMMANDS) {
   210→   210→                s.on(cmd) { args -> main.post { ackCommand(cmd, args) } }
   211→   211→            }
   212→   212→
   213→   213→            // ==== FA-7: جلسة التحكم + أوامر اللمس/النص/الأزرار + فتح التطبيقات ====
   214→   214→            s.on(AgentBridge.EV_CTL_START) { args ->
   215→   215→                val data = args.firstOrNull() as? JSONObject ?: return@on
   216→   216→                main.post { handleCtlStart(data) }
   217→   217→            }
   218→   218→            s.on(AgentBridge.EV_CTL_STOP) { _ ->
   219→   219→                main.post { endControlSession("stopped") }
   220→   220→            }
   221→   221→            s.on(AgentBridge.EV_TOUCH) { args ->
   222→   222→                val data = args.firstOrNull() as? JSONObject ?: return@on
   223→   223→                // اللمس على المعالج الرئيسي — dispatchGesture يتطلبه عمليًا للحساسية القصوى
   224→   224→                main.post { handleTouch(data) }
   225→   225→            }
   226→   226→            s.on(AgentBridge.EV_TEXT) { args ->
   227→   227→                val data = args.firstOrNull() as? JSONObject ?: return@on
   228→   228→                main.post { handleText(data) }
   229→   229→            }
   230→   230→            s.on(AgentBridge.EV_KEY) { args ->
   231→   231→                val data = args.firstOrNull() as? JSONObject ?: return@on
   232→   232→                main.post { handleKey(data) }
   233→   233→            }
   234→   234→            s.on(AgentBridge.EV_APP) { args ->
   235→   235→                val data = args.firstOrNull() as? JSONObject ?: return@on
   236→   236→                main.post { handleAppOpen(data) }
   237→   237→            }
   238→   238→
   239→   239→            // عروض نقل الملفات (عقد §8): إشعار نظام + بث إلى الواجهة
   240→   240→            s.on(AgentBridge.EV_TX_OFFER) { args ->
   241→   241→                val data = args.firstOrNull() as? JSONObject ?: return@on

   244→   244→            // أوامر الكمبيوتر على مهام التنزيل — تُبث للواجهة فقط (المحرك يعيش فيها)

   246→   246→                s.on(ev) { args ->

   249→   249→                        MainActivity.pushDataEvent("txcmd", mapOf(
   250→   250→                            "event" to ev,
   251→   251→                            "id" to data.optString("id"),
   252→   252→                        ))
   253→   253→                    }
   254→   254→                }
   255→   255→            }
   256→   256→
   257→   257→            s.connect()
   258→   258→            startWatchdog()
   259→   259→        } catch (e: Exception) {

   359→   359→            String.format("%.1fGB", sizeMb / 1024)
   360→   360→        } else String.format("%.1fMB", sizeMb)
   361→   361→        notifyTx(
   362→   362→            id = id,
   363→   363→            title = "ملف وارد من ${data.optString("from", "الكمبيوتر")}",
   364→   364→            body = "${data.optString("fileName")} — $sizeLabel",

   366→   366→        )

   369→   369→    /** إشعار نقل: قبول يفتح التطبيق، رفض يُرسل مباشرة عبر القناة الدائمة. */
   370→   370→    private fun notifyTx(id: Int, title: String, body: String, offerId: String?) {
   371→   371→        try {
   372→   372→            val contentIntent = PendingIntent.getActivity(
   373→   373→                this, id,
   374→   374→                Intent(this, MainActivity::class.java).apply {
   375→   375→                    flags = Intent.FLAG_ACTIVITY_SINGLE_TOP
   376→   376→                    offerId?.let { putExtra("nexus.offerId", it) }
   377→   377→                },
   378→   378→                PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE,
   379→   379→            )
   380→   380→
   381→   381→            val builder = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
   382→   382→                Notification.Builder(this, TX_CHANNEL_ID)
   383→   383→            } else {
   384→   384→                @Suppress("DEPRECATION")
   385→   385→                Notification.Builder(this)
   386→   386→            }
   387→   387→            builder.setContentTitle(title)
   388→   388→                .setContentText(body)
   389→   389→                .setSmallIcon(android.R.drawable.stat_sys_download)
   390→   390→                .setContentIntent(contentIntent)
   391→   391→                .setAutoCancel(true)
   392→   392→
   393→   393→            if (offerId != null) {
   394→   394→                // قبول → افتح التطبيق ليبدأ التنزيل بالمحرك الكامل
   395→   395→                builder.addAction(
   396→   396→                    0, "قبول",
   397→   397→                    contentIntent,
   398→   398→                )
   399→   399→                // رفض → الخدمة ترفض مباشرة عبر القناة الدائمة
   400→   400→                val rejectIntent = PendingIntent.getService(
   401→   401→                    this, id + 1,
   402→   402→                    Intent(this, AgentForegroundService::class.java).apply {
   403→   403→                        putExtra("nexus.action", "reject")
   404→   404→                        putExtra("nexus.offerId", offerId)
   405→   405→                    },
   406→   406→                    PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE,
   407→   407→                )
   408→   408→                builder.addAction(0, "رفض", rejectIntent)
   409→   409→            }
   410→   410→
   411→   411→            (getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager)
   412→   412→                .notify(id, builder.build())
   413→   413→        } catch (_: Exception) {
   414→   414→        }
   415→   415→    }

   417→   417→    private fun handleServiceAction(intent: Intent?) {
   418→   418→        when (intent?.getStringExtra("nexus.action")) {
   419→   419→            "reject" -> {
   420→   420→                val offerId = intent.getStringExtra("nexus.offerId") ?: return
   421→   421→                try {
   422→   422→                    socket?.emit(AgentBridge.EV_TX_REJECT, JSONObject().put("id", offerId))
   423→   423→                    (getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager)
   424→   424→                        .cancel(offerId.hashCode() and 0x7fffffff)
   425→   425→                } catch (_: Exception) {
   426→   426→                }
   427→   427→            }
   428→   428→            "ctl-allow" -> {

   548→   548→    private fun handleTouch(data: JSONObject) {
   549→   549→        if (!ctlActive) return // بوابة الجلسة — لا جلسة، لا حقن
   550→   550→        val svc = NexusAccessibilityService.instance
   551→   551→        val op = data.optString("op")
   552→   552→        val seq = data.optInt("seq", 0)
   553→   553→        if (svc == null) {
   554→   554→            emitTouchAck(seq, op, false, "accessibility-off")
   555→   555→            return
   556→   556→        }
   557→   557→        val x = data.optDouble("x", 0.0).toFloat()
   558→   558→        val y = data.optDouble("y", 0.0).toFloat()
   559→   559→        when (op) {
   560→   560→            "tap" -> svc.tap(x, y, seq)
   561→   561→            "long" -> svc.longPress(x, y, data.optLong("durationMs", 700), seq)
   562→   562→            "swipe" -> svc.swipe(
   563→   563→                x, y,
   564→   564→                data.optDouble("x2", x.toDouble()).toFloat(),
   565→   565→                data.optDouble("y2", y.toDouble()).toFloat(),
   566→   566→                data.optLong("durationMs", 300), "swipe", seq,
   567→   567→            )
   568→   568→            "scroll" -> svc.scroll(
   569→   569→                x, y,
   570→   570→                data.optDouble("x2", x.toDouble()).toFloat() - x,
   571→   571→                data.optDouble("y2", y.toDouble()).toFloat() - y,
   572→   572→                data.optLong("durationMs", 300), seq,
   573→   573→            )
   574→   574→            "down" -> svc.touchDown(x, y)
   575→   575→            "move" -> svc.touchMove(x, y)
   576→   576→            "up" -> svc.touchUp(x, y)
   577→   577→            else -> if (seq > 0) emitTouchAck(seq, op, false, "unknown-op")
   578→   578→        }
   579→   579→    }
