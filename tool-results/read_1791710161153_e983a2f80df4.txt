    65→        fun stashOffer(o: JSONObject) {
    66→            synchronized(offerBacklog) {
    67→                offerBacklog.addLast(JSONObject(o.toString()))
    68→                while (offerBacklog.size > BACKLOG_MAX) offerBacklog.removeFirst()
    69→            }
    70→        }

    73→            synchronized(offerBacklog) {
    74→                val out = ArrayList<JSONObject>(offerBacklog)
    75→                offerBacklog.clear()
    76→                return out
    77→            }
    78→        }

   184→    private fun connect() {
   185→        val url = if (mode == "gateway") "http://$host:$port" else "http://$host:$bridgePort"
   186→        try {
   187→            val opts = IO.Options()
   188→            opts.reconnection = true
   189→            opts.reconnectionDelay = 2_000
   190→            opts.reconnectionDelayMax = 15_000
   191→            opts.timeout = 10_000
   192→            if (mode == "gateway") {
   193→                // نفس نمط عميل الويب: Caddy يوجّه ?XTransformPort إلى منفذ الجسر
   194→                opts.query = "XTransformPort=$bridgePort"
   195→            }
   196→
   197→            val s = IO.socket(url, opts)
   198→            socket = s
   199→
   200→            s.on(Socket.EVENT_CONNECT, onConnect)
   201→            s.on(Socket.EVENT_DISCONNECT) {
   202→                setState("offline")
   203→                // انقطاع القناة = نهاية الجلسة دائمًا (من يتحكم لم يعد متصلًا)
   204→                main.post { endControlSession("disconnected") }
   205→            }
   206→            s.on(Socket.EVENT_CONNECT_ERROR) { setState("reconnecting") }
   207→
   208→            // أوامر الكمبيوتر → الهاتف: إقرار الاستلام (التنفيذ الكامل في مرحلة الحماية)
   209→            for (cmd in AgentBridge.PC_COMMANDS) {
   210→                s.on(cmd) { args -> main.post { ackCommand(cmd, args) } }
   211→            }
   212→
   213→            // ==== FA-7: جلسة التحكم + أوامر اللمس/النص/الأزرار + فتح التطبيقات ====
   214→            s.on(AgentBridge.EV_CTL_START) { args ->
   215→                val data = args.firstOrNull() as? JSONObject ?: return@on
   216→                main.post { handleCtlStart(data) }
   217→            }
   218→            s.on(AgentBridge.EV_CTL_STOP) { _ ->
   219→                main.post { endControlSession("stopped") }
   220→            }
   221→            s.on(AgentBridge.EV_TOUCH) { args ->
   222→                val data = args.firstOrNull() as? JSONObject ?: return@on
   223→                // اللمس على المعالج الرئيسي — dispatchGesture يتطلبه عمليًا للحساسية القصوى
   224→                main.post { handleTouch(data) }
   225→            }
   226→            s.on(AgentBridge.EV_TEXT) { args ->
   227→                val data = args.firstOrNull() as? JSONObject ?: return@on
   228→                main.post { handleText(data) }
   229→            }
   230→            s.on(AgentBridge.EV_KEY) { args ->
   231→                val data = args.firstOrNull() as? JSONObject ?: return@on
   232→                main.post { handleKey(data) }
   233→            }
   234→            s.on(AgentBridge.EV_APP) { args ->
   235→                val data = args.firstOrNull() as? JSONObject ?: return@on
   236→                main.post { handleAppOpen(data) }
   237→            }
   238→
   239→            // عروض نقل الملفات (عقد §8): إشعار نظام + بث إلى الواجهة
   240→            s.on(AgentBridge.EV_TX_OFFER) { args ->
   241→                val data = args.firstOrNull() as? JSONObject ?: return@on
   242→                main.post { handleTxOffer(data) }
   243→            }
   244→            // أوامر الكمبيوتر على مهام التنزيل — تُبث للواجهة فقط (المحرك يعيش فيها)
   245→            for (ev in listOf(AgentBridge.EV_TX_PAUSE, AgentBridge.EV_TX_RESUME, AgentBridge.EV_TX_CANCEL)) {
   246→                s.on(ev) { args ->
   247→                    val data = args.firstOrNull() as? JSONObject ?: return@on
   248→                    main.post {
   249→                        MainActivity.pushDataEvent("txcmd", mapOf(
   250→                            "event" to ev,
   251→                            "id" to data.optString("id"),
   252→                        ))
   253→                    }
   254→                }
   255→            }
   256→
   257→            s.connect()
   258→            startWatchdog()
   259→        } catch (e: Exception) {
   260→            setState("offline")
   261→            android.util.Log.e(TAG, "connect failed", e)
   262→        }
   263→    }

   269→    private fun connectProd() {
   270→        val prodPort = AgentBridge.PROD_PORT
   271→        val url = if (mode == "gateway") "http://$host:$port" else "http://$host:$prodPort"
   272→        try {
   273→            val opts = IO.Options()
   274→            opts.reconnection = true
   275→            opts.reconnectionDelay = 2_000
   276→            opts.reconnectionDelayMax = 15_000
   277→            opts.timeout = 10_000
   278→            if (mode == "gateway") {
   279→                opts.query = "XTransformPort=$prodPort"
   280→            }
   281→            val s = IO.socket(url, opts)
   282→            prodSocket = s
   283→
   284→            s.on(Socket.EVENT_CONNECT) {
   285→                android.util.Log.i(TAG, "prod bridge online")
   286→            }
   287→            s.on(AgentBridge.EV_CLIP_NEW) { args ->
   288→                val data = args.firstOrNull() as? JSONObject ?: return@on
   289→                main.post {
   290→                    MainActivity.pushDataEvent("clip", mapOf(
   291→                        "id" to data.optString("id"),
   292→                        "deviceId" to data.optString("deviceId"),
   293→                        "deviceName" to data.optString("deviceName"),
   294→                        "kind" to data.optString("kind"),
   295→                        "content" to data.optString("content"),
   296→                        "fileName" to if (data.has("fileName")) data.optString("fileName") else null,
   297→                        "size" to data.optLong("size"),
   298→                        "mode" to data.optString("mode"),
   299→                        "createdAt" to data.optLong("createdAt"),
   300→                    ))
   301→                    if (!uiAlive && data.optString("kind") == "text") {
   302→                        notifyTx(
   303→                            id = 7300 + (data.optString("id").hashCode() and 0x7fffffff) % 300,
   304→                            title = "نص من ${data.optString("deviceName", "الكمبيوتر")}",
   305→                            body = data.optString("content").take(160),
   306→                            offerId = null,
   307→                        )
   308→                    }
   309→                }
   310→            }
   311→            s.on(AgentBridge.EV_HAND_NEW) { args ->
   312→                val data = args.firstOrNull() as? JSONObject ?: return@on
   313→                main.post {
   314→                    MainActivity.pushDataEvent("hand", mapOf(
   315→                        "id" to data.optString("id"),
   316→                        "deviceId" to data.optString("deviceId"),
   317→                        "deviceName" to data.optString("deviceName"),
   318→                        "kind" to data.optString("kind"),
   319→                        "title" to data.optString("title"),
   320→                        "url" to data.optString("url"),
   321→                        "status" to data.optString("status"),
   322→                        "createdAt" to data.optLong("createdAt"),
   323→                    ))
   324→                    if (!uiAlive) {
   325→                        notifyTx(
   326→                            id = 7600 + (data.optString("id").hashCode() and 0x7fffffff) % 300,
   327→                            title = "رابط من ${data.optString("deviceName", "الكمبيوتر")}",
   328→                            body = data.optString("title").ifEmpty { data.optString("url") }.take(160),
   329→                            offerId = null,
   330→                        )
   331→                    }
   332→                }
   333→            }
   334→            s.connect()
   335→        } catch (e: Exception) {
   336→            android.util.Log.e(TAG, "prod connect failed", e)
   337→        }
   338→    }

   344→    private fun handleTxOffer(data: JSONObject) {
   345→        stashOffer(data)
   346→        MainActivity.pushDataEvent("tx", mapOf(
   347→            "id" to data.optString("id"),
   348→            "fileName" to data.optString("fileName"),
   349→            "size" to data.optLong("size"),
   350→            "mime" to if (data.has("mime")) data.optString("mime") else null,
   351→            "from" to data.optString("from"),
   352→            "srcDeviceId" to data.optString("srcDeviceId"),
   353→            "srcVPath" to data.optString("srcVPath"),
   354→            "at" to data.optLong("at"),
   355→        ))
   356→        val id = data.optString("id").hashCode() and 0x7fffffff
   357→        val sizeMb = data.optLong("size") / (1024.0 * 1024.0)
   358→        val sizeLabel = if (sizeMb >= 1024) {
   359→            String.format("%.1fGB", sizeMb / 1024)
   360→        } else String.format("%.1fMB", sizeMb)
   361→        notifyTx(
   362→            id = id,
   363→            title = "ملف وارد من ${data.optString("from", "الكمبيوتر")}",
   364→            body = "${data.optString("fileName")} — $sizeLabel",
   365→            offerId = data.optString("id"),
   366→        )
   367→    }

   369→    /** إشعار نقل: قبول يفتح التطبيق، رفض يُرسل مباشرة عبر القناة الدائمة. */
   370→    private fun notifyTx(id: Int, title: String, body: String, offerId: String?) {
   371→        try {
   372→            val contentIntent = PendingIntent.getActivity(
   373→                this, id,
   374→                Intent(this, MainActivity::class.java).apply {
   375→                    flags = Intent.FLAG_ACTIVITY_SINGLE_TOP
   376→                    offerId?.let { putExtra("nexus.offerId", it) }
   377→                },
   378→                PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE,
   379→            )
   380→
   381→            val builder = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
   382→                Notification.Builder(this, TX_CHANNEL_ID)
   383→            } else {
   384→                @Suppress("DEPRECATION")
   385→                Notification.Builder(this)
   386→            }
   387→            builder.setContentTitle(title)
   388→                .setContentText(body)
   389→                .setSmallIcon(android.R.drawable.stat_sys_download)
   390→                .setContentIntent(contentIntent)
   391→                .setAutoCancel(true)
   392→
   393→            if (offerId != null) {
   394→                // قبول → افتح التطبيق ليبدأ التنزيل بالمحرك الكامل
   395→                builder.addAction(
   396→                    0, "قبول",
   397→                    contentIntent,
   398→                )
   399→                // رفض → الخدمة ترفض مباشرة عبر القناة الدائمة
   400→                val rejectIntent = PendingIntent.getService(
   401→                    this, id + 1,
   402→                    Intent(this, AgentForegroundService::class.java).apply {
   403→                        putExtra("nexus.action", "reject")
   404→                        putExtra("nexus.offerId", offerId)
   405→                    },
   406→                    PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE,
   407→                )
   408→                builder.addAction(0, "رفض", rejectIntent)
   409→            }
   410→
   411→            (getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager)
   412→                .notify(id, builder.build())
   413→        } catch (_: Exception) {
   414→        }
   415→    }
   416→
   417→    private fun handleServiceAction(intent: Intent?) {
   418→        when (intent?.getStringExtra("nexus.action")) {
   419→            "reject" -> {
   420→                val offerId = intent.getStringExtra("nexus.offerId") ?: return
   421→                try {
   422→                    socket?.emit(AgentBridge.EV_TX_REJECT, JSONObject().put("id", offerId))
   423→                    (getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager)
   424→                        .cancel(offerId.hashCode() and 0x7fffffff)
   425→                } catch (_: Exception) {
   426→                }
   427→            }
   428→            "ctl-allow" -> {
   429→                val from = ctlPendingFrom
   430→                val name = ctlPendingName
   431→                ctlPendingFrom = null
   432→                ctlPendingName = null
   433→                cancelCtlNotification()
   434→                if (from != null) acceptControl(from, name ?: "الكمبيوتر")
   435→            }
   436→            "ctl-deny" -> {
   437→                val from = ctlPendingFrom
   438→                val name = ctlPendingName
   439→                ctlPendingFrom = null
   440→                ctlPendingName = null
   441→                cancelCtlNotification()
   442→                if (from != null) {
   443→                    pushCtlToUi("state", mapOf("active" to false, "controller" to from, "reason" to "denied"))
   444→                    emitCtlState(false, "denied", from, name)
   445→                }
   446→            }
   447→        }
   448→    }

   460→    private fun emitCtlState(active: Boolean, reason: String, controller: String?, name: String?) {
   461→        try {
   462→            val o = JSONObject()
   463→                .put("active", active)
   464→                .put("reason", reason)
   465→                .put("at", System.currentTimeMillis())
   466→            if (!controller.isNullOrEmpty()) o.put("controller", controller)
   467→            if (!name.isNullOrEmpty()) o.put("name", name)
   468→            socket?.emit(AgentBridge.EV_CTL_STATE, o)
   469→        } catch (_: Exception) {
   470→        }
   471→    }

   473→    private fun handleCtlStart(data: JSONObject) {
   474→        val from = data.optString("from")
   475→        val name = data.optString("name").ifEmpty { "الكمبيوتر" }
   476→        if (from.isEmpty()) return
   477→        if (ctlActive) {
   478→            // الجلسة الواحدة فقط في الوقت نفسه — بلّغ بأن الخط مشغول
   479→            emitCtlState(true, "busy", ctlController, ctlName)
   480→            return
   481→        }
   482→        if (ctlRequireConfirm) {
   483→            ctlPendingFrom = from
   484→            ctlPendingName = name
   485→            notifyCtlRequest(name)
   486→            pushCtlToUi("start-request", mapOf("controller" to from, "name" to name))
   487→            // لا حالة بعد — القرار قادم من الإشعار أو حوار الواجهة
   488→        } else {
   489→            acceptControl(from, name)
   490→        }
   491→    }
   492→
   493→    private fun acceptControl(from: String, name: String) {
   494→        ctlActive = true
   495→        ctlController = from
   496→        ctlName = name
   497→        cancelCtlNotification()
   498→        pushCtlToUi("state", mapOf("active" to true, "controller" to from, "name" to name, "reason" to "confirmed"))
   499→        emitCtlState(true, "confirmed", from, name)
   500→        emitDevState()
   501→        android.util.Log.i(TAG, "control session started by $from ($name)")
   502→    }
   503→
   504→    private fun endControlSession(reason: String) {
   505→        if (!ctlActive) return
   506→        val c = ctlController
   507→        val n = ctlName
   508→        ctlActive = false
   509→        ctlController = null
   510→        ctlName = null
   511→        cancelCtlNotification()
   512→        pushCtlToUi("state", mapOf("active" to false, "controller" to (c ?: ""), "reason" to reason))
   513→        emitCtlState(false, reason, c, n)
   514→        emitDevState()
   515→        android.util.Log.i(TAG, "control session ended: $reason")
   516→    }

   518→    /** قرار حوار الواجهة (MethodChannel — MainActivity). */
   519→    fun decideControl(allow: Boolean) {
   520→        val from = ctlPendingFrom
   521→        val name = ctlPendingName
   522→        ctlPendingFrom = null
   523→        ctlPendingName = null
   524→        cancelCtlNotification()
   525→        if (allow && from != null) {
   526→            acceptControl(from, name ?: "الكمبيوتر")
   527→        } else if (from != null) {
   528→            pushCtlToUi("state", mapOf("active" to false, "controller" to from, "reason" to "denied"))
   529→            emitCtlState(false, "denied", from, name)
   530→        }
   531→    }

   539→        try {
   540→            val o = JSONObject().put("op", op).put("ok", ok)
   541→            if (seq > 0) o.put("seq", seq)
   542→            if (err != null) o.put("err", err)

   548→    private fun handleTouch(data: JSONObject) {
   549→        if (!ctlActive) return // بوابة الجلسة — لا جلسة، لا حقن
   550→        val svc = NexusAccessibilityService.instance
   551→        val op = data.optString("op")
   552→        val seq = data.optInt("seq", 0)
   553→        if (svc == null) {
   554→            emitTouchAck(seq, op, false, "accessibility-off")
   555→            return
   556→        }
   557→        val x = data.optDouble("x", 0.0).toFloat()
   558→        val y = data.optDouble("y", 0.0).toFloat()
   559→        when (op) {
   560→            "tap" -> svc.tap(x, y, seq)
   561→            "long" -> svc.longPress(x, y, data.optLong("durationMs", 700), seq)
   562→            "swipe" -> svc.swipe(
   563→                x, y,
   564→                data.optDouble("x2", x.toDouble()).toFloat(),
   565→                data.optDouble("y2", y.toDouble()).toFloat(),
   566→                data.optLong("durationMs", 300), "swipe", seq,
   567→            )
   568→            "scroll" -> svc.scroll(
   569→                x, y,
   570→                data.optDouble("x2", x.toDouble()).toFloat() - x,
   571→                data.optDouble("y2", y.toDouble()).toFloat() - y,
   572→                data.optLong("durationMs", 300), seq,
   573→            )
   574→            "down" -> svc.touchDown(x, y)
   575→            "move" -> svc.touchMove(x, y)
   576→            "up" -> svc.touchUp(x, y)
   577→            else -> if (seq > 0) emitTouchAck(seq, op, false, "unknown-op")
   578→        }
   579→    }

   602→    private fun handleAppOpen(data: JSONObject) {
   603→        val action = data.optString("action", "open")
   604→        val pkg = data.optString("pkg")
   605→        if (action != "open" || pkg.isEmpty()) {

   608→        }
   609→        try {
   610→            val intent = packageManager.getLaunchIntentForPackage(pkg)
   611→            if (intent != null) {
   612→                intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
   613→                startActivity(intent)
   614→                socket?.emit(
   615→                    AgentBridge.EV_DEV_ACKED,
   616→                    JSONObject().put("text", "فُتح تطبيق ${data.optString("name", pkg)} من الكمبيوتر"),
   617→                )
   618→            } else {
   619→                socket?.emit(
   620→                    AgentBridge.EV_DEV_ACKED,
   621→                    JSONObject().put("text", "تعذّر فتح ${data.optString("name", pkg)} — التطبيق غير مثبت على الهاتف"),
   622→                )
   623→            }
   624→        } catch (e: Exception) {
   625→            socket?.emit(
   626→                AgentBridge.EV_DEV_ACKED,
   627→                JSONObject().put("text", "خطأ في فتح $pkg: ${e.message?.take(60) ?: "مجهول"}"),
   628→            )
   629→        }
   630→    }
