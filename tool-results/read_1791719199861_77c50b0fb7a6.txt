   341→    private fun handleIncoming(intent: Intent?) {
   342→        if (intent == null) return
   343→        when {
   344→            // من إشعار "قبول" → أخبر الواجهة لتبدأ التنزيل
   345→            intent.hasExtra("nexus.offerId") -> {
   346→                val offerId = intent.getStringExtra("nexus.offerId")
   347→                if (offerId != null) {
   348→                    pushDataEvent("txaction", mapOf("id" to offerId, "action" to "accept"))
   349→                }
   350→            }
   351→            // قائمة المشاركة: نص أو ملف(ات) من أي تطبيق (البند 4)
   352→            intent.action == Intent.ACTION_SEND || intent.action == Intent.ACTION_SEND_MULTIPLE -> {
   353→                Thread { stashFromShareIntent(intent); pushDataEvent("share", emptyMap()) }.start()
   354→            }
   355→            // زر "إيقاف البث" من إشعار الوسائط (FA-8)
   356→            intent.hasExtra("nexus.mediaStop") -> {
   357→                pushDataEvent("mediastop", emptyMap())
   358→            }
   359→            // FA-9: ودجت الإرسال السريع — إرسال حافظة الهاتف إلى الكمبيوتر
   360→            intent.getStringExtra(NexusWidgets.OPEN_QUICKSEND) == NexusWidgets.OPEN_QUICKSEND_VAL -> {
   361→                pushDataEvent("quicksend", mapOf("from" to "widget"))
   362→            }
   363→        }
   364→    }

   366→    private fun stashFromShareIntent(intent: Intent) {
   367→        val item = HashMap<String, Any?>()
   368→        val files = ArrayList<HashMap<String, Any?>>()
   369→        when (intent.action) {
   370→            Intent.ACTION_SEND -> {
   371→                (intent.getStringExtra(Intent.EXTRA_TEXT))?.let { item["text"] = it }
   372→                @Suppress("DEPRECATION")
   373→                val uri = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
   374→                    intent.getParcelableExtra(Intent.EXTRA_STREAM, Uri::class.java)
   375→                } else {
   376→                    intent.getParcelableExtra(Intent.EXTRA_STREAM)
   377→                }
   378→                if (uri != null) copyUriToCache(uri)?.let { files.add(it) }
   379→            }
   380→            Intent.ACTION_SEND_MULTIPLE -> {
   381→                @Suppress("DEPRECATION")
   382→                @Suppress("UNCHECKED_CAST")
   383→                val uris = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
   384→                    intent.getParcelableArrayListExtra(Intent.EXTRA_STREAM, Uri::class.java)
   385→                } else {
   386→                    intent.getParcelableArrayListExtra<Uri>(Intent.EXTRA_STREAM)
   387→                }
   388→                uris?.forEach { u -> copyUriToCache(u)?.let { files.add(it) } }
   389→            }
   390→        }
   391→        if (item["text"] != null || files.isNotEmpty()) {
   392→            item["files"] = files
   393→            stashShare(item)
   394→        }
   395→    }

   397→    private fun copyUriToCache(uri: Uri): HashMap<String, Any?>? {
   398→        return try {
   399→            val nameCol = arrayOf(MediaStore.MediaColumns.DISPLAY_NAME, MediaStore.MediaColumns.SIZE, MediaStore.MediaColumns.MIME_TYPE)
   400→            var name = "file_${System.currentTimeMillis()}"
   401→            var size = 0L
   402→            var mime = "application/octet-stream"
   403→            contentResolver.query(uri, nameCol, null, null, null)?.use { c ->
   404→                if (c.moveToFirst()) {
   405→                    c.getString(0)?.let { name = it }
   406→                    size = c.getLong(1)
   407→                    c.getString(2)?.let { mime = it }
   408→                }
   409→            }
   410→            val dir = File(cacheDir, "nexus-share").apply { mkdirs() }
   411→            val dst = File(dir, "${System.currentTimeMillis()}_$name")
   412→            contentResolver.openInputStream(uri)?.use { input ->
   413→                dst.outputStream().use { output -> input.copyTo(output) }
   414→            } ?: return null
   415→            if (size <= 0L) size = dst.length()
   416→            hashMapOf(
   417→                "path" to dst.absolutePath,
   418→                "name" to name,
   419→                "size" to size,
   420→                "mime" to mime,
   421→            )
   422→        } catch (_: Exception) {
   423→            null
   424→        }
   425→    }

   431→    private fun publishDownload(tempPath: String, displayName: String, mime: String): String? {
   432→        val temp = File(tempPath)
   433→        if (!temp.exists() || temp.length() == 0L) return null
   434→        return try {
   435→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   436→                val resolver = contentResolver
   437→                val values = ContentValues().apply {
   438→                    put(MediaStore.Downloads.DISPLAY_NAME, displayName)
   439→                    put(MediaStore.Downloads.MIME_TYPE, mime)
   440→                    put(MediaStore.Downloads.RELATIVE_PATH, Environment.DIRECTORY_DOWNLOADS + "/NexusLink")
   441→                    put(MediaStore.Downloads.IS_PENDING, 1)
   442→                }
   443→                val uri = resolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values) ?: return null
   444→                resolver.openOutputStream(uri)?.use { out ->
   445→                    temp.inputStream().use { it.copyTo(out) }
   446→                } ?: return null
   447→                values.clear()
   448→                values.put(MediaStore.Downloads.IS_PENDING, 0)
   449→                resolver.update(uri, values, null, null)
   450→                "${Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS)}/NexusLink/$displayName"
   451→            } else {
   452→                val dir = File(Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS), "NexusLink")
   453→                dir.mkdirs()
   454→                var dst = File(dir, displayName)
   455→                var i = 1
   456→                while (dst.exists()) {
   457→                    val dot = displayName.lastIndexOf('.')
   458→                    val stem = if (dot > 0) displayName.substring(0, dot) else displayName
   459→                    val ext = if (dot > 0) displayName.substring(dot) else ""
   460→                    dst = File(dir, "$stem ($i)$ext")
   461→                    i++
   462→                }
   463→                temp.copyTo(dst)
   464→                dst.absolutePath
   465→            }
   466→        } catch (_: Exception) {
   467→            null
   468→        }
   469→    }

   471→    private fun notifyDownloadDone(name: String, path: String) {
   472→        try {
   473→            val nm = getSystemService(NOTIFICATION_SERVICE) as NotificationManager
   474→            val builder = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
   475→                android.app.Notification.Builder(this, AgentForegroundService.TX_CHANNEL_ID)
   476→            } else {
   477→                @Suppress("DEPRECATION")
   478→                android.app.Notification.Builder(this)
   479→            }
   480→            builder.setContentTitle("اكتمل استلام الملف")
   481→                .setContentText("$name — حُفظ في Download/NexusLink")
   482→                .setSmallIcon(android.R.drawable.stat_sys_download_done)
   483→                .setAutoCancel(true)
   484→            val id = ("done" + path).hashCode() and 0x7fffffff
   485→            nm.notify(id, builder.build())
   486→        } catch (_: Exception) {
   487→        }
   488→    }

   502→    private fun queryNewPhotos(since: Long, limit: Int): List<HashMap<String, Any?>> {
   503→        if (!hasPhotosPermission()) return emptyList()
   504→        val out = ArrayList<HashMap<String, Any?>>()
   505→        try {
   506→            val collection = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   507→                MediaStore.Images.Media.getContentUri(MediaStore.VOLUME_EXTERNAL)
   508→            } else {
   509→                MediaStore.Images.Media.EXTERNAL_CONTENT_URI
   510→            }
   511→            val proj = arrayOf(
   512→                MediaStore.Images.Media._ID,
   513→                MediaStore.Images.Media.DISPLAY_NAME,
   514→                MediaStore.Images.Media.SIZE,
   515→                MediaStore.Images.Media.DATE_ADDED,
   516→                MediaStore.Images.Media.MIME_TYPE,
   517→                MediaStore.Images.Media.DATA,
   518→            )
   519→            contentResolver.query(collection, proj, null, null, "${MediaStore.Images.Media.DATE_ADDED} ASC")?.use { c ->
   520→                while (c.moveToNext() && out.size < limit) {
   521→                    val dateAdded = c.getLong(3)
   522→                    val dataPath = c.getString(5) ?: ""
   523→                    // DCIM أو Pictures فقط (المراقبة المطلوبة) — ودعم <Q عبر المسار
   524→                    val inScope = dataPath.contains("/DCIM/") || dataPath.contains("/Pictures/") ||
   525→                        dataPath.contains("\\DCIM\\") || dataPath.contains("\\Pictures\\")
   526→                    if (!inScope) continue
   527→                    if (dateAdded <= since) continue
   528→                    try {
   529→                        val uri = Uri.withAppendedPath(collection, c.getString(0))
   530→                        val dir = File(cacheDir, "nexus-photos").apply { mkdirs() }
   531→                        val dst = File(dir, "${dateAdded}_${c.getString(1) ?: "photo.jpg"}")
   532→                        if (!dst.exists() || dst.length() == 0L) {
   533→                            contentResolver.openInputStream(uri)?.use { input ->
   534→                                dst.outputStream().use { input.copyTo(it) }
   535→                            } ?: continue
   536→                        }
   537→                        out.add(
   538→                            hashMapOf(
   539→                                "path" to dst.absolutePath,
   540→                                "name" to (c.getString(1) ?: "photo.jpg"),
   541→                                "size" to (if (c.getLong(2) > 0) c.getLong(2) else dst.length()),
   542→                                "dateAdded" to dateAdded,
   543→                                "mime" to (c.getString(4) ?: "image/jpeg"),
   544→                            ),
   545→                        )
   546→                    } catch (_: Exception) {
   547→                    }
   548→                }
   549→            }
   550→        } catch (_: Exception) {
   551→        }
   552→        return out
   553→    }

   559→    private fun readDeviceInfo(): Map<String, Any?> {
   560→        val battery = try {
   561→            val bm = getSystemService(BATTERY_SERVICE) as BatteryManager
   562→            bm.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY)
   563→        } catch (_: Exception) {
   564→            100
   565→        }
   566→        val deviceName = try {
   567→            Settings.Global.getString(contentResolver, "device_name")
   568→        } catch (_: Exception) {
   569→            null
   570→        } ?: Build.DEVICE
   571→        return mapOf(
   572→            "androidId" to (
   573→                Settings.Secure.getString(contentResolver, Settings.Secure.ANDROID_ID) ?: ""
   574→                ),
   575→            "model" to Build.MODEL,
   576→            "manufacturer" to Build.MANUFACTURER,
   577→            "deviceName" to deviceName,
   578→            "os" to "Android ${Build.VERSION.RELEASE} (SDK ${Build.VERSION.SDK_INT})",
   579→            "battery" to battery,
   580→        )
   581→    }

   583→    private fun guessMime(name: String): String {
   584→        val ext = name.substringAfterLast('.', "").lowercase()
   585→        return mapOf(
   586→            "jpg" to "image/jpeg", "jpeg" to "image/jpeg", "png" to "image/png", "gif" to "image/gif", "webp" to "image/webp",
   587→            "mp4" to "video/mp4", "mkv" to "video/x-matroska", "webm" to "video/webm", "mov" to "video/quicktime",
   588→            "mp3" to "audio/mpeg", "wav" to "audio/wav", "m4a" to "audio/mp4", "flac" to "audio/flac", "ogg" to "audio/ogg",
   589→            "pdf" to "application/pdf", "apk" to "application/vnd.android.package-archive", "zip" to "application/zip",
   590→            "txt" to "text/plain", "md" to "text/plain", "csv" to "text/plain", "json" to "text/plain",
   591→        )[ext] ?: "application/octet-stream"
   592→    }
